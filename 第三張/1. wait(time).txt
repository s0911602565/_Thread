目的 : 單獨使用 wait(time ) 時間到自動釋放鎖


class A {

}

public class MethodFactory2 {

    static private Object lock = new Object();
    static public Runnable r = new Runnable() {

        public void run() {
            try {
                synchronized (lock) {
                    System.out.println(System.currentTimeMillis());
                    lock.wait(3000);
                    System.out.println(System.currentTimeMillis());

                }
            } catch (Exception e) {
                e.printStackTrace();
            }
        }
    };

    public static void main(String[] args) throws Exception {
        Thread t = new Thread(r);
        t.start();
    }
}
